 10 #define NUMBER_OF_THREADS 8
 11  
 12 volatile int entering[NUMBER_OF_THREADS];
 13 volatile int number[NUMBER_OF_THREADS];
 14 int shared_counter=0;
 15  
 16 int max(volatile int* array, int size) {
 17     int result = *(array);
 18     for(int i=0; i<size; i++) {
 19         if(result<*(array+i)){
 20             result = *(array+i);
 21         }
 22     }
 23     return result;
 24 }
 25  
 26 void lock(int i) {
 27     entering[i] = true;
 28     number[i] = 1 + max(&number[0], sizeof(number)/sizeof(int));
 29     entering[i] = false;
 30     for (int j = 0; j < NUMBER_OF_THREADS; j++) {
 31         while (entering[j]);
 32         while (number[j] != 0 &&
 33               (number[j] < number[i] || (number[j] == number[i] && j < i)));
 34     }
 35 }
 36  
 37 void unlock(int i) {
 38     number[i] = 0;
 39 }
 40  
 41 void *bakery(void *ptr_tid)
 42 {
 43     int tid = *(int*)ptr_tid;
 44     free(ptr_tid);
 45     while(true) {
 46         lock(tid);
 47         if (shared_counter >= INT_MAX-1) {
 48             unlock(tid);
 49             break;
 50         }
 51         shared_counter++;
 52         printf("Hey, Critical section of thread %d here!, Counter is at %d\n", tid, shared_counter);
 53         unlock(tid);
 54     }
 55     pthread_exit(NULL);
 56 }
 57  
 58 int main(int argc, char *argv[])
 59 {
 60     pthread_t threads[NUMBER_OF_THREADS];
 61  
 62     for(int i=0; i<NUMBER_OF_THREADS; i++)
 63     {
 64         int *ptr_i = malloc(sizeof(int));
 65         *ptr_i = i;
 66         printf("Main here. Creating thread %d\n", i);
 67         if (pthread_create(&threads[i], NULL, &bakery, ptr_i) != 0)
 68             err(errno, "Failed creating thread");
 69     }
 70     for(int i=0; i<NUMBER_OF_THREADS; i++)
 71     {
 72         pthread_join(threads[i], NULL);
 73     }
 74     exit(0);
 75 }
